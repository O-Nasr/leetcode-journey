public class Solution {
    public bool IsValidSudoku(char[][] board) {
        HashSet<string> sudokuHashed = new HashSet<string>();

        for (int i = 0; i < 9; i++)
        {
            for (int j = 0; j < 9; j++)
            {
                if (board[i][j] != '.')
                {
                    if (!sudokuHashed.Add("R"+i+"-"+board[i][j]))
                    {
                        return false;
                    }
                    if (!sudokuHashed.Add("C"+j+"-"+board[i][j]))
                    {
                        return false;
                    }
                    if (!sudokuHashed.Add("B"+i/3+j/3+"-"+board[i][j]))
                    {
                        return false;
                    }
                }
            }
        }
        
        return true;
    }
}
