public class Solution {
    public int ReverseBits(int n) {
        string binary = Convert.ToString(n, 2).PadLeft(32, '0');
        char[] chars = binary.ToCharArray();
        Array.Reverse(chars);
        return Convert.ToInt32(new string(chars), 2);
    }
}
